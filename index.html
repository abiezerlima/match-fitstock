async function carregarEmpresas() { if (!currentOppId) { log("ERRO: currentOppId não foi definido."); setStatus("Oportunidade não identificada.", "error"); return; } log("carregarEmpresas() executado com oppId = " + currentOppId); setStatus("Buscando empresas compatíveis...", "loading"); // Formato exato esperado pelo SDK para REST API Functions const reqData = { arguments: JSON.stringify({ oppId: currentOppId }) }; log("Executando ZOHO.CRM.FUNCTIONS.execute('" + FUNCTION_API_NAME + "')"); log("Payload: " + JSON.stringify(reqData)); try { const response = await ZOHO.CRM.FUNCTIONS.execute(FUNCTION_API_NAME, reqData); log("Resposta da FUNCTIONS.execute:"); log(JSON.stringify(response, null, 2)); if (!response) { throw new Error("A função não retornou resposta."); } let output = response.output; if (typeof output === "string") { try { output = JSON.parse(output); } catch (parseError) { log("Output não é string JSON. Mantendo valor bruto."); } } if (output && output.status === "error") { throw new Error(output.message || "A função Deluge retornou um erro."); } if (output && Array.isArray(output.empresas)) { renderResults(output.empresas); setStatus("Busca concluída com sucesso.", "success"); } else { renderResults([]); setStatus("Nenhuma empresa encontrada.", "success"); } } catch (error) { console.error("Erro em FUNCTIONS.execute:", error); log("ERRO em FUNCTIONS.execute: " + JSON.stringify(error)); setStatus("Erro ao chamar a função.", "error"); } }
